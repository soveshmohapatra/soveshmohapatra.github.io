% Section header: letter-spaced gray uppercase + hairline rule (full width).
\newcommand{\reservelines}[1]{\par\begingroup\dimen0=\pagegoal\advance\dimen0 by -\pagetotal\ifdim\dimen0<#1\newpage\fi\endgroup}
\newcommand{\cvsection}[1]{%
  \reservelines{7\baselineskip}%
  \par\addvspace{5pt}\noindent%
  {\large\color{headgray}\textls[140]{\MakeUppercase{#1}}}%
  \par\nointerlineskip\vspace{5pt}\noindent{\color{rulegray}\hrule height 0.6pt}\par\addvspace{5pt}%
}

% Lightweight subheader (e.g. publication categories), aligned to content col.
\newcommand{\cvsubsection}[1]{%
  \reservelines{12\baselineskip}%
  \par\addvspace{6pt}\noindent\textbf{#1}\par\addvspace{3pt}%
}

% Dated entry: right-aligned date in left column, body in content column.
% \dated{date}{body}
\newcommand{\dated}[2]{%
  \par\addvspace{3pt}\noindent
  \begin{minipage}[t]{\datewidth}\RaggedLeft\small #1\strut\end{minipage}%
  \hspace{\datesep}%
  \begin{minipage}[t]{\contentwidth}#2\strut\end{minipage}\par%
}

% Dated entry variant with a left-aligned date.
% \datedleft{date}{body}
\newcommand{\datedleft}[2]{%
  \par\vspace{3pt}\noindent
  \begin{minipage}[t]{\datewidth}\RaggedRight\small #1\strut\end{minipage}%
  \hspace{\datesep}%
  \begin{minipage}[t]{\contentwidth}#2\strut\end{minipage}\par%
}

% Compact dated entry with a left-aligned date and tighter date-to-text gap.

\newlength{\compactsep}\setlength{\compactsep}{0.08in}
\newlength{\compactcontentwidth}
\setlength{\compactcontentwidth}{\dimexpr\textwidth-\compactdatewidth-\compactsep\relax}
\newcommand{\datedcompact}[2]{%
  \par\vspace{3pt}\noindent
  \begin{minipage}[t]{\compactdatewidth}\RaggedRight\small #1\strut\end{minipage}%
  \hspace{\compactsep}%
  \begin{minipage}[t]{\compactcontentwidth}#2\strut\end{minipage}\par%
}
\newcommand{\datedresearch}[2]{%
  \par\vspace{3pt}\noindent
  \begin{minipage}[t]{\compactdatewidth}\RaggedRight\small #1\strut\end{minipage}%
  \hspace{\compactsep}%
  \begin{minipage}[t]{\compactcontentwidth}#2\strut\end{minipage}\par%
}

\newenvironment{researchbullets}{%
  \begin{list}{\textbullet}{%
    \setlength{\topsep}{0pt}\setlength{\partopsep}{0pt}%
    \setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}%
    \setlength{\leftmargin}{1.2em}\setlength{\labelwidth}{0.8em}%
    \setlength{\labelsep}{0.4em}}}%
  {\end{list}\vspace{-6pt}}

% A bold "group title" sitting in the content column with no date

  \par\vspace{6pt}\noindent\hspace*{\contentindent}%
  \begin{minipage}[t]{\contentwidth}\textbf{#1}\strut\end{minipage}\par\vspace{1pt}%
}

% Compact variants for presentations, keeping titles close to the year column.

\setlength{\talkcontentwidth}{\dimexpr\textwidth-\talkdatewidth-\talksep\relax}

\newcommand{\talktitle}[1]{%
  \par\addvspace{1pt}\noindent\hspace*{\talkindent}%
  \begin{minipage}[t]{\talkcontentwidth}\textbf{#1}\strut\end{minipage}\par\nobreak\vspace{1pt}%
}
\newcommand{\talkdated}[2]{%
  \par\vspace{3pt}\noindent
  \begin{minipage}[t]{\talkdatewidth}\RaggedRight\small #1\strut\end{minipage}%

  \begin{minipage}[t]{\talkcontentwidth}#2\strut\end{minipage}\par%
}

% Compact bullet list (base "list" env; no enumitem needed).

    \setlength{\leftmargin}{1.2em}\setlength{\labelwidth}{0.8em}%
    \setlength{\labelsep}{0.4em}}}%
